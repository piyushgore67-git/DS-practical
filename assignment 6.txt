#include <stdio.h>
#include <stdlib.h>
struct node
{
int roll;
struct node *left, *right;
};
struct node* createNode(int roll)
{
struct node *newNode;
newNode = (struct node*)malloc(sizeof(struct node));
newNode->roll = roll;
newNode->left = NULL;
newNode->right = NULL;
return newNode;
}
struct node* insert(struct node *root, int roll)
{
if (root == NULL)
return createNode(roll);
if (roll < root->roll)
root->left = insert(root->left, roll);
else if (roll > root->roll)
root->right = insert(root->right, roll);
else printf("Duplicate roll number not allowed!\n");
return root;
}
struct node* search(struct node *root, int roll)
{
if (root == NULL || root->roll == roll)
return root;
if (roll < root->roll)
return search(root->left, roll);
return search(root->right, roll);
}
struct node* findMin(struct node *root)
{
while (root != NULL && root->left != NULL)
root = root->left;
return root;
}
struct node* deleteNode(struct node *root, int roll)
{
struct node *temp;
if (root == NULL)
{
printf("Roll number not found!\n");
return root;
}
if (roll < root->roll)
root->left = deleteNode(root->left, roll);
else if (roll > root->roll)
root->right = deleteNode(root->right, roll);
else
{
if (root->left == NULL)
{
temp = root->right;
free(root);
return temp;
}
else if (root->right == NULL)
{ temp = root->left;
free(root);
return temp;
}
temp = findMin(root->right);
root->roll = temp->roll;
root->right = deleteNode(root->right, temp->roll);
}
return root;
}
void inorder(struct node *root)
{
if (root != NULL)
{
inorder(root->left);
printf("%d ", root->roll);
inorder(root->right);
}
}
void preorder(struct node *root)
{
if (root != NULL)
{
printf("%d ", root->roll);
preorder(root->left);
preorder(root->right);
}
}
void postorder(struct node *root)
{
if (root != NULL)
{
postorder(root->left);
postorder(root->right);
printf("%d ", root->roll);
}
}
int main()
{
struct node *root = NULL;
struct node *result;
int choice, roll; while (1)
{
printf("\n\n--- BST MENU ---\n");
printf("1. Insert a student roll number\n");
printf("2. Delete a student roll number\n");
printf("3. Search a student roll number\n");
printf("4. Display Inorder Traversal\n");
printf("5. Display Preorder Traversal\n");
printf("6. Display Postorder Traversal\n");
printf("7. Exit\n");
printf("Enter your choice: ");
scanf("%d", &choice);
switch (choice)
{
case 1:
printf("Enter roll number: ");
scanf("%d", &roll);
if (search(root, roll) != NULL)
printf("Duplicate roll number not allowed!\n");
else
{
root = insert(root, roll);
printf("Roll number inserted successfully.\n");
}
break;
case 2:
printf("Enter roll number to delete: ");
scanf("%d", &roll);
if (search(root, roll) == NULL)
printf("Roll number not found!\n");
else
{
root = deleteNode(root, roll);
printf("Roll number deleted successfully.\n");
}
break;
case 3:
printf("Enter roll number to search: ");
scanf("%d", &roll);
result = search(root, roll);
if (result != NULL)
printf("Roll number %d found.\n", roll);
else printf("Roll number not found.\n");
break;
case 4:
printf("Inorder Traversal: ");
inorder(root);
printf("\n");
break;
case 5:
printf("Preorder Traversal: ");
preorder(root);
printf("\n");
break;
case 6:
printf("Postorder Traversal: ");
postorder(root);
printf("\n");
break;
case 7:
printf("Exiting program.\n");
return 0;
default:
printf("Invalid choice!\n");
}
}
}
